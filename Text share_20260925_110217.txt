/* =========================================================
   STORAGE
========================================================= */

const STORAGE_KEY =
    "BTA_PUBLISHING_DASHBOARD_V1";


/* =========================================================
   DEFAULT DATA
========================================================= */

let data =
    JSON.parse(
        localStorage.getItem(STORAGE_KEY)
    ) || createDefaultData();


function createDefaultData(){

    const chapters = [];

    for(let i = 1; i <= 100; i++){

        chapters.push({

            number:i,

            title:"Chapter " + i,

            platforms:{
                webnovel:{
                    done:false,
                    date:""
                },

                royalroad:{
                    done:false,
                    date:""
                },

                patreon:{
                    done:false,
                    date:""
                }
            }

        });

    }

    return {
        chapters:chapters
    };
}


function save(){

    localStorage.setItem(
        STORAGE_KEY,
        JSON.stringify(data)
    );

    updateStats();

}


/* =========================================================
   DATE SYSTEM
========================================================= */

/*
    First Sunday:
    27 September 2026

    First batch:
    Chapters 1–5

    Then:
    Monday-Friday = 1 chapter/day
    Saturday-Sunday = rest
*/

const FIRST_SUNDAY =
    new Date(2026,8,27);


function royalRoadDate(chapterNumber){

    /*
        Chapter 1-5:
        27 September 2026

        Chapter 6 onward:
        Monday-Friday releases.
    */

    if(chapterNumber <= 5){

        return new Date(
            2026,
            8,
            27
        );

    }

    let date =
        new Date(
            FIRST_SUNDAY
        );

    let remaining =
        chapterNumber - 5;

    while(remaining > 0){

        date.setDate(
            date.getDate() + 1
        );

        const day =
            date.getDay();

        if(day >= 1 && day <= 5){

            remaining--;

        }

    }

    return date;

}


function formatDate(date){

    return date.toLocaleDateString(
        "en-IN",
        {
            weekday:"long",
            day:"numeric",
            month:"long",
            year:"numeric"
        }
    );

}


/* =========================================================
   PLATFORM
========================================================= */

let currentPlatform = "all";


document
.querySelectorAll(".tab")
.forEach(tab => {

    tab.addEventListener(
        "click",
        () => {

            document
            .querySelectorAll(".tab")
            .forEach(t =>
                t.classList.remove("active")
            );

            tab.classList.add("active");

            currentPlatform =
                tab.dataset.platform;

            updatePlatformHeader();

            render();

        }
    );

});


function platformLabel(){

    if(currentPlatform === "webnovel")
        return "Webnovel";

    if(currentPlatform === "royalroad")
        return "Royal Road";

    if(currentPlatform === "patreon")
        return "Patreon";

    return "All Platforms";

}


/* =========================================================
   RENDER
========================================================= */

function render(){

    const list =
        document.getElementById("list");

    const search =
        document
        .getElementById("search")
        .value
        .toLowerCase();

    const filter =
        document
        .getElementById("filter")
        .value;

    list.innerHTML = "";


    data.chapters.forEach(
        (chapter,index) => {

            const status =
                chapterStatus(chapter);

            const searchable =
                (
                    chapter.number +
                    " " +
                    chapter.title
                ).toLowerCase();

            if(
                search &&
                !searchable.includes(search)
            ){
                return;
            }


            if(
                filter !== "all" &&
                status !== filter
            ){
                return;
            }


            const box =
                document.createElement("div");

            box.className =
                "chapter";

            box.id =
                "chapter-" +
                chapter.number;


            const head =
                document.createElement("div");

            head.className =
                "chapter-head";


            const expand =
                document.createElement("button");

            expand.className =
                "expand";

            expand.textContent =
                "+";


            expand.onclick = () => {

                box.classList.toggle("open");

                expand.textContent =
                    box.classList.contains("open")
                    ? "−"
                    : "+";

            };


            const number =
                document.createElement("div");

            number.className =
                "chapter-number";

            number.textContent =
                "Chapter " +
                chapter.number;


            const info =
                document.createElement("div");

            info.className =
                "chapter-info";


            const title =
                document.createElement("div");

            title.className =
                "chapter-title";

            title.textContent =
                chapter.title;


            const date =
                document.createElement("div");

            date.className =
                "chapter-date";

            const rrDate =
                royalRoadDate(
                    chapter.number
                );

            date.textContent =
                "Royal Road: " +
                formatDate(rrDate);


            info.appendChild(title);
            info.appendChild(date);


            const statusBadge =
                document.createElement("div");

            statusBadge.className =
                "status " +
                (
                    status === "complete"
                    ? "complete"
                    : status === "partial"
                    ? "partial"
                    : ""
                );

            statusBadge.textContent =
                status === "complete"
                ? "Complete"
                : status === "partial"
                ? "Partial"
                : "Pending";


            head.appendChild(expand);
            head.appendChild(number);
            head.appendChild(info);
            head.appendChild(statusBadge);


            const body =
                document.createElement("div");

            body.className =
                "chapter-body";


            const titleInput =
                document.createElement("input");

            titleInput.className =
                "title-input";

            titleInput.value =
                chapter.title;

            titleInput.placeholder =
                "Chapter title";

            titleInput.oninput = () => {

                chapter.title =
                    titleInput.value;

                save();

            };


            body.appendChild(titleInput);


            const grid =
                document.createElement("div");

            grid.className =
                "platform-grid";


            createPlatformCard(
                grid,
                chapter,
                "webnovel",
                "Webnovel"
            );

            createPlatformCard(
                grid,
                chapter,
                "royalroad",
                "Royal Road"
            );

            createPlatformCard(
                grid,
                chapter,
                "patreon",
                "Patreon"
            );


            body.appendChild(grid);

            box.appendChild(head);
            box.appendChild(body);

            list.appendChild(box);

        }
    );

}


function createPlatformCard(
    grid,
    chapter,
    key,
    label
){

    const card =
        document.createElement("div");

    card.className =
        "platform-card";


    const title =
        document.createElement("div");

    title.className =
        "platform-card-title";

    title.textContent =
        label;


    const row =
        document.createElement("label");

    row.className =
        "check-row";


    const checkbox =
        document.createElement("input");

    checkbox.type =
        "checkbox";

    checkbox.checked =
        chapter.platforms[key].done;


    checkbox.onchange = () => {

        chapter.platforms[key].done =
            checkbox.checked;

        save();
        render();

    };


    const text =
        document.createElement("span");

    text.textContent =
        checkbox.checked
        ? "Published / Uploaded"
        : "Not completed";


    row.appendChild(checkbox);
    row.appendChild(text);


    const date =
        document.createElement("input");

    date.type =
        "datetime-local";

    date.className =
        "date-input";

    date.value =
        chapter.platforms[key].date;


    /*
        Royal Road default schedule
    */

    if(
        key === "royalroad" &&
        !chapter.platforms[key].date
    ){

        const d =
            royalRoadDate(
                chapter.number
            );

        const y =
            d.getFullYear();

        const m =
            String(
                d.getMonth()+1
            ).padStart(2,"0");

        const day =
            String(
                d.getDate()
            ).padStart(2,"0");

        date.value =
            `${y}-${m}-${day}T15:14`;

    }


    date.onchange = () => {

        chapter.platforms[key].date =
            date.value;

        save();

    };


    card.appendChild(title);
    card.appendChild(row);
    card.appendChild(date);

    grid.appendChild(card);

}


/* =========================================================
   STATUS
========================================================= */

function chapterStatus(chapter){

    const values =
        Object.values(
            chapter.platforms
        );

    const completed =
        values.filter(
            x => x.done
        ).length;


    if(completed === 3)
        return "complete";

    if(completed > 0)
        return "partial";

    return "pending";

}


/* =========================================================
   STATS
========================================================= */

function updateStats(){

    const total =
        data.chapters.length;

    let tasks = 0;

    let totalTasks =
        total * 3;

    let completeChapters = 0;


    data.chapters.forEach(
        chapter => {

            const values =
                Object.values(
                    chapter.platforms
                );

            const done =
                values.filter(
                    x => x.done
                ).length;

            tasks += done;

            if(done === 3)
                completeChapters++;

        }
    );


    const percent =
        totalTasks
        ? Math.round(
            (tasks / totalTasks) * 100
        )
        : 0;


    document
    .getElementById("totalChapters")
    .textContent =
        total;


    document
    .getElementById("fullyPublished")
    .textContent =
        completeChapters;


    document
    .getElementById("completedTasks")
    .textContent =
        tasks +
        " / " +
        totalTasks;


    document
    .getElementById("overallProgress")
    .textContent =
        percent + "%";


    updatePlatformHeader();

}


function updatePlatformHeader(){

    const name =
        platformLabel();

    document
    .getElementById("platformName")
    .textContent =
        name;


    let completed = 0;
    let total = 0;


    data.chapters.forEach(
        chapter => {

            if(currentPlatform === "all"){

                Object.values(
                    chapter.platforms
                ).forEach(platform => {

                    total++;

                    if(platform.done)
                        completed++;

                });

            } else {

                total++;

                if(
                    chapter
                    .platforms[
                        currentPlatform
                    ].done
                ){
                    completed++;
                }

            }

        }
    );


    const percent =
        total
        ? Math.round(
            (completed / total) * 100
        )
        : 0;


    document
    .getElementById("platformPercent")
    .textContent =
        percent + "%";


    document
    .getElementById("platformBar")
    .style.width =
        percent + "%";

}


/* =========================================================
   ADD CHAPTER
========================================================= */

function addChapter(){

    const next =
        data.chapters.length + 1;


    data.chapters.push({

        number:next,

        title:
            "Chapter " + next,

        platforms:{

            webnovel:{
                done:false,
                date:""
            },

            royalroad:{
                done:false,
                date:""
            },

            patreon:{
                done:false,
                date:""
            }

        }

    });


    save();
    render();


    setTimeout(() => {

        const element =
            document.getElementById(
                "chapter-" + next
            );

        if(element){

            element.scrollIntoView({
                behavior:"smooth",
                block:"center"
            });

            const button =
                element.querySelector(
                    ".expand"
                );

            if(button)
                button.click();

        }

    },100);

}


/* =========================================================
   EXPORT
========================================================= */

function exportData(){

    const json =
        JSON.stringify(
            data,
            null,
            2
        );


    const blob =
        new Blob(
            [json],
            {
                type:"application/json"
            }
        );


    const url =
        URL.createObjectURL(blob);


    const a =
        document.createElement("a");

    a.href =
        url;

    a.download =
        "before-the-last-archive-publishing-data.json";

    a.click();


    URL.revokeObjectURL(url);

}


/* =========================================================
   RESET
========================================================= */

function resetAll(){

    const confirmReset =
        confirm(
            "This will delete your saved publishing progress. Continue?"
        );

    if(!confirmReset)
        return;


    localStorage.removeItem(
        STORAGE_KEY
    );


    data =
        createDefaultData();


    render();
    updateStats();

}


/* =========================================================
   INITIALIZE
========================================================= */

render();
updateStats();
